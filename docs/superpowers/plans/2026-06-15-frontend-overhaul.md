# Frontend Overhaul Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the blank Next.js template with a functional industrial-dark MES UI covering Dashboard, Production Orders, and Delivery Orders — all browsable with mock data while the real API is still being built.

**Architecture:** Next.js App Router with a fixed sidebar layout. A `lib/api.ts` data layer has a `USE_MOCK` constant at the top: `true` returns hardcoded data, `false` calls the real FastAPI backend at `http://localhost:8000`. Pages are client components that call `lib/api.ts` on mount. No server-side data fetching needed yet — mock mode works without a running backend.

**Tech Stack:** Next.js 16, React 19, Tailwind CSS v4, shadcn/ui (new-york), lucide-react

```typescript
// frontend/lib/api.ts
import type {
  Article,
  ProductionOrder,
  DeliveryOrder,
  Machine,
  OrderFormData,
} from './types'

const USE_MOCK = true
const BASE_URL = 'http://localhost:8000'

// ── Mock data ─────────────────────────────────────────────────────────────────

const MOCK_ARTICLES: Article[] = [
  { id: 1, name: 'Stahlträger A400', description: 'Standardstahlträger 400mm', price: 299.99 },
  { id: 2, name: 'Aluminiumplatte B200', description: 'Aluminiumblech 200×200 mm', price: 89.50 },
  { id: 3, name: 'Verbundrohr C100', description: 'Verbundrohr DN100', price: 45.00 },
  { id: 4, name: 'Kupferleitung D50', description: 'Kupferrohr 50mm', price: 32.75 },
]

let MOCK_PRODUCTION_ORDERS: ProductionOrder[] = [
  { id: 1, quantity: 50, status: 'completed', article_id: 1, article: MOCK_ARTICLES[0] },
  { id: 2, quantity: 25, status: 'in_progress', article_id: 2, article: MOCK_ARTICLES[1] },
  { id: 3, quantity: 100, status: 'pending', article_id: 1, article: MOCK_ARTICLES[0] },
  { id: 4, quantity: 10, status: 'error', article_id: 3, article: MOCK_ARTICLES[2] },
  { id: 5, quantity: 75, status: 'pending', article_id: 2, article: MOCK_ARTICLES[1] },
]

let MOCK_DELIVERY_ORDERS: DeliveryOrder[] = [
  { id: 1, quantity: 30, status: 'completed', article_id: 2, article: MOCK_ARTICLES[1] },
  { id: 2, quantity: 15, status: 'pending', article_id: 1, article: MOCK_ARTICLES[0] },
  { id: 3, quantity: 8, status: 'in_progress', article_id: 3, article: MOCK_ARTICLES[2] },
]

const MOCK_MACHINES: Machine[] = [
  { id: 1, name: 'Fräsmaschine 1', type: 'CNC Fräse', status: 'online' },
  { id: 2, name: 'Schweißroboter A', type: 'Schweißanlage', status: 'online' },
  { id: 3, name: 'Drehmaschine 2', type: 'CNC Drehmaschine', status: 'offline' },
  { id: 4, name: 'Bohrwerk B', type: 'Bohranlage', status: 'online' },
]

let mockIdCounter = 100

// ── Helpers ───────────────────────────────────────────────────────────────────

async function apiFetch<T>(path: string, options?: RequestInit): Promise<T> {
  const res = await fetch(`${BASE_URL}${path}`, {
    headers: { 'Content-Type': 'application/json' },
    ...options,
  })
  if (!res.ok) throw new Error(`API error ${res.status}: ${path}`)
  return res.json()
}

function resolveArticle(article_id: number): Article | undefined {
  return MOCK_ARTICLES.find(a => a.id === article_id)
}

// ── API object ────────────────────────────────────────────────────────────────

export const api = {
  articles: {
    list: async (): Promise<Article[]> => {
      if (USE_MOCK) return MOCK_ARTICLES
      return apiFetch('/article/')
    },
  },

  productionOrders: {
    list: async (): Promise<ProductionOrder[]> => {
      if (USE_MOCK) return MOCK_PRODUCTION_ORDERS
      return apiFetch('/productionorder/')
    },
    create: async (data: OrderFormData): Promise<ProductionOrder> => {
      if (USE_MOCK) {
        const order: ProductionOrder = {
          id: ++mockIdCounter,
          ...data,
          article: resolveArticle(data.article_id),
        }
        MOCK_PRODUCTION_ORDERS = [...MOCK_PRODUCTION_ORDERS, order]
        return order
      }
      return apiFetch('/productionorder/', {
        method: 'POST',
        body: JSON.stringify(data),
      })
    },
    update: async (id: number, data: OrderFormData): Promise<ProductionOrder> => {
      if (USE_MOCK) {
        MOCK_PRODUCTION_ORDERS = MOCK_PRODUCTION_ORDERS.map(o =>
          o.id === id
            ? { ...o, ...data, article: resolveArticle(data.article_id) }
            : o
        )
        return MOCK_PRODUCTION_ORDERS.find(o => o.id === id)!
      }
      return apiFetch(`/productionorder/${id}`, {
        method: 'PATCH',
        body: JSON.stringify(data),
      })
    },
    delete: async (id: number): Promise<void> => {
      if (USE_MOCK) {
        MOCK_PRODUCTION_ORDERS = MOCK_PRODUCTION_ORDERS.filter(o => o.id !== id)
        return
      }
      await apiFetch(`/productionorder/${id}`, { method: 'DELETE' })
    },
  },

  deliveryOrders: {
    list: async (): Promise<DeliveryOrder[]> => {
      if (USE_MOCK) return MOCK_DELIVERY_ORDERS
      return apiFetch('/deliveryorder/')
    },
    create: async (data: OrderFormData): Promise<DeliveryOrder> => {
      if (USE_MOCK) {
        const order: DeliveryOrder = {
          id: ++mockIdCounter,
          ...data,
          article: resolveArticle(data.article_id),
        }
        MOCK_DELIVERY_ORDERS = [...MOCK_DELIVERY_ORDERS, order]
        return order
      }
      return apiFetch('/deliveryorder/', {
        method: 'POST',
        body: JSON.stringify(data),
      })
    },
    update: async (id: number, data: OrderFormData): Promise<DeliveryOrder> => {
      if (USE_MOCK) {
        MOCK_DELIVERY_ORDERS = MOCK_DELIVERY_ORDERS.map(o =>
          o.id === id
            ? { ...o, ...data, article: resolveArticle(data.article_id) }
            : o
        )
        return MOCK_DELIVERY_ORDERS.find(o => o.id === id)!
      }
      return apiFetch(`/deliveryorder/${id}`, {
        method: 'PATCH',
        body: JSON.stringify(data),
      })
    },
    delete: async (id: number): Promise<void> => {
      if (USE_MOCK) {
        MOCK_DELIVERY_ORDERS = MOCK_DELIVERY_ORDERS.filter(o => o.id !== id)
        return
      }
      await apiFetch(`/deliveryorder/${id}`, { method: 'DELETE' })
    },
  },

  machines: {
    list: async (): Promise<Machine[]> => {
      if (USE_MOCK) return MOCK_MACHINES
      return apiFetch('/machine/')
```

```tsx
// frontend/app/dashboard/page.tsx
'use client'

import { useState, useEffect } from 'react'
import { api } from '@/lib/api'
import type { ProductionOrder, DeliveryOrder, Machine } from '@/lib/types'
import { StatCard } from '@/components/stat-card'
import { StatusBadge } from '@/components/status-badge'
import { Factory, Truck, Activity, CheckCircle2, Cpu } from 'lucide-react'

export default function DashboardPage() {
  const [productionOrders, setProductionOrders] = useState<ProductionOrder[]>([])
  const [deliveryOrders, setDeliveryOrders] = useState<DeliveryOrder[]>([])
  const [machines, setMachines] = useState<Machine[]>([])

  useEffect(() => {
    Promise.all([
      api.productionOrders.list(),
      api.deliveryOrders.list(),
      api.machines.list(),
    ]).then(([prod, del, mach]) => {
      setProductionOrders(prod)
      setDeliveryOrders(del)
      setMachines(mach)
    })
  }, [])

  const inProgress =
    productionOrders.filter(o => o.status === 'in_progress').length +
    deliveryOrders.filter(o => o.status === 'in_progress').length

  const completed =
    productionOrders.filter(o => o.status === 'completed').length +
    deliveryOrders.filter(o => o.status === 'completed').length

  return (
    <div className="p-8 space-y-8">
      <div>
        <h1 className="text-2xl font-bold text-foreground">Dashboard</h1>
        <p className="text-muted-foreground text-sm mt-1">Fertigungsübersicht</p>
      </div>

      {/* Summary stats */}
      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
        <StatCard
          label="Fertigungsaufträge"
          value={productionOrders.length}
          icon={Factory}
          accent="orange"
        />
        <StatCard
          label="Lieferaufträge"
          value={deliveryOrders.length}
          icon={Truck}
          accent="blue"
        />
        <StatCard
          label="In Bearbeitung"
          value={inProgress}
          icon={Activity}
          accent="amber"
        />
        <StatCard
          label="Abgeschlossen"
          value={completed}
          icon={CheckCircle2}
          accent="green"
        />
      </div>

      {/* Machine status */}
      <section className="space-y-3">
        <h2 className="text-muted-foreground font-semibold text-xs uppercase tracking-wider">
          Maschinen
        </h2>
        <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
          {machines.map(machine => (
            <div
              key={machine.id}
              className="bg-card border border-border rounded-lg p-4"
            >
              <div className="flex items-start justify-between">
                <Cpu size={18} className="text-muted-foreground mt-0.5" />
                <StatusBadge status={machine.status} />
              </div>
              <p className="text-foreground font-medium mt-3 text-sm">{machine.name}</p>
              <p className="text-muted-foreground text-xs">{machine.type}</p>
            </div>
          ))}
        </div>
      </section>

      {/* Recent production orders */}
      <section className="space-y-3">
        <h2 className="text-muted-foreground font-semibold text-xs uppercase tracking-wider">
          Aktuelle Fertigungsaufträge
        </h2>
        <div className="bg-card border border-border rounded-lg divide-y divide-border">
          {productionOrders.length === 0 ? (
            <p className="text-muted-foreground text-sm text-center py-8">
              Keine Fertigungsaufträge
            </p>
          ) : (
            productionOrders.slice(0, 5).map(order => (
              <div
                key={order.id}
                className="px-5 py-3.5 flex items-center justify-between"
              >
                <div className="flex items-center gap-3">
                  <span className="text-muted-foreground font-mono text-xs">
                    #{order.id}
                  </span>
                  <span className="text-foreground text-sm">
                    {order.article?.name ?? `Artikel #${order.article_id}`}
                  </span>
                </div>
                <div className="flex items-center gap-4">
                  <span className="text-muted-foreground text-sm">
                    {order.quantity} Stk.
                  </span>
                  <StatusBadge status={order.status} />
                </div>
              </div>
            ))
          )}
        </div>
      </section>
    </div>
  )
}
```

```tsx
} from '@/components/ui/dialog'
import { Input } from '@/components/ui/input'
import { Label } from '@/components/ui/label'
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from '@/components/ui/select'
import {
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
} from '@/components/ui/table'
import {
  DropdownMenu,
  DropdownMenuContent,
  DropdownMenuItem,
  DropdownMenuTrigger,
} from '@/components/ui/dropdown-menu'

type Order = ProductionOrder | DeliveryOrder

interface OrdersSectionProps {
  type: 'production' | 'delivery'
  title: string
}

const ORDER_STATUSES: OrderStatus[] = ['pending', 'in_progress', 'error', 'completed']

const STATUS_LABELS: Record<OrderStatus, string> = {
  pending: 'Ausstehend',
  in_progress: 'In Bearbeitung',
  error: 'Fehler',
  completed: 'Abgeschlossen',
}

export function OrdersSection({ type, title }: OrdersSectionProps) {
  const [orders, setOrders] = useState<Order[]>([])
  const [articles, setArticles] = useState<Article[]>([])
  const [loading, setLoading] = useState(true)
  const [dialogOpen, setDialogOpen] = useState(false)
  const [editingOrder, setEditingOrder] = useState<Order | null>(null)
  const [form, setForm] = useState<OrderFormData>({
    quantity: 1,
    status: 'pending',
    article_id: 0,
  })

  const ordersApi = type === 'production' ? api.productionOrders : api.deliveryOrders

  useEffect(() => {
    Promise.all([ordersApi.list(), api.articles.list()]).then(
      ([ordersData, articlesData]) => {
        setOrders(ordersData)
        setArticles(articlesData)
        if (articlesData.length > 0) {
          setForm(f => ({ ...f, article_id: articlesData[0].id }))
        }
        setLoading(false)
      }
    )
  }, [type])

  const openCreate = () => {
    setEditingOrder(null)
    setForm({
      quantity: 1,
      status: 'pending',
      article_id: articles[0]?.id ?? 0,
    })
    setDialogOpen(true)
  }

  const openEdit = (order: Order) => {
    setEditingOrder(order)
    setForm({
      quantity: order.quantity,
      status: order.status,
      article_id: order.article_id,
    })
    setDialogOpen(true)
  }

  const handleSubmit = async () => {
    if (editingOrder) {
      const updated = await ordersApi.update(editingOrder.id, form)
      setOrders(prev => prev.map(o => (o.id === editingOrder.id ? updated : o)))
    } else {
      const created = await ordersApi.create(form)
      setOrders(prev => [...prev, created])
    }
    setDialogOpen(false)
  }

  const handleDelete = async (id: number) => {
    await ordersApi.delete(id)
    setOrders(prev => prev.filter(o => o.id !== id))
  }

  const counts = {
    pending: orders.filter(o => o.status === 'pending').length,
    in_progress: orders.filter(o => o.status === 'in_progress').length,
    error: orders.filter(o => o.status === 'error').length,
    completed: orders.filter(o => o.status === 'completed').length,
  }

  return (
    <div className="p-8 space-y-8">
      {/* Page header */}
      <div className="flex items-center justify-between">
        <h1 className="text-2xl font-bold text-foreground">{title}</h1>
        <Button
          onClick={openCreate}
          className="bg-primary text-primary-foreground hover:bg-primary/90"
        >
          <Plus size={16} className="mr-2" />
          Neuer Auftrag
        </Button>
      </div>

      {/* Stat cards */}
      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
        <StatCard label="Ausstehend" value={counts.pending} icon={Clock} accent="amber" />
        <StatCard label="In Bearbeitung" value={counts.in_progress} icon={Activity} accent="blue" />
        <StatCard label="Fehler" value={counts.error} icon={AlertCircle} accent="red" />
        <StatCard label="Abgeschlossen" value={counts.completed} icon={CheckCircle2} accent="green" />
      </div>

      {/* Orders table */}
      <div className="bg-card border border-border rounded-lg overflow-hidden">
        <Table>
          <TableHeader>
            <TableRow className="border-border hover:bg-transparent">
              <TableHead className="text-muted-foreground">ID</TableHead>
              <TableHead className="text-muted-foreground">Artikel</TableHead>
              <TableHead className="text-muted-foreground">Menge</TableHead>
              <TableHead className="text-muted-foreground">Status</TableHead>
              <TableHead className="text-muted-foreground text-right">Aktionen</TableHead>
            </TableRow>
          </TableHeader>
          <TableBody>
            {loading ? (
              <TableRow>
                <TableCell
                  colSpan={5}
                  className="text-center text-muted-foreground py-12"
                >
                  Lade Aufträge…
                </TableCell>
              </TableRow>
            ) : orders.length === 0 ? (
              <TableRow>
                <TableCell
                  colSpan={5}
                  className="text-center text-muted-foreground py-12"
                >
                  Keine Aufträge vorhanden. Erstelle einen neuen Auftrag.
                </TableCell>
              </TableRow>
            ) : (
              orders.map(order => (
                <TableRow
                  key={order.id}
                  className="border-border hover:bg-secondary/50"
                >
                  <TableCell className="text-muted-foreground font-mono text-xs">
                    #{order.id}
                  </TableCell>
                  <TableCell className="text-foreground">
                    {order.article?.name ?? `Artikel #${order.article_id}`}
                  </TableCell>
                  <TableCell className="text-foreground">{order.quantity}</TableCell>
                  <TableCell>
                    <StatusBadge status={order.status} />
                  </TableCell>
                  <TableCell className="text-right">
                    <DropdownMenu>
                      <DropdownMenuTrigger asChild>
                        <Button
                          variant="ghost"
                          size="icon"
                          className="text-muted-foreground hover:text-foreground"
                        >
                          <MoreHorizontal size={16} />
                        </Button>
                      </DropdownMenuTrigger>
                      <DropdownMenuContent align="end" className="bg-card border-border">
                        <DropdownMenuItem
                          onClick={() => openEdit(order)}
                          className="cursor-pointer"
                        >
                          <Pencil size={14} className="mr-2" />
                          Bearbeiten
                        </DropdownMenuItem>
                        <DropdownMenuItem
                          onClick={() => handleDelete(order.id)}
                          className="text-destructive cursor-pointer"
                        >
                          <Trash2 size={14} className="mr-2" />
                          Löschen
                        </DropdownMenuItem>
                      </DropdownMenuContent>
                    </DropdownMenu>
                  </TableCell>
                </TableRow>
              ))
            )}
          </TableBody>
        </Table>
      </div>

      {/* Create / Edit dialog */}
      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
        <DialogContent className="bg-card border-border">
          <DialogHeader>
            <DialogTitle>
              {editingOrder ? 'Auftrag bearbeiten' : 'Neuer Auftrag'}
            </DialogTitle>
          </DialogHeader>

          <div className="space-y-4 py-4">
            <div className="space-y-2">
              <Label>Artikel</Label>
              <Select
                value={String(form.article_id)}
                onValueChange={v =>
                  setForm(f => ({ ...f, article_id: Number(v) }))
                }
              >
                <SelectTrigger className="bg-secondary border-border">
                  <SelectValue />
                </SelectTrigger>
                <SelectContent className="bg-card border-border">
                  {articles.map(a => (
                    <SelectItem key={a.id} value={String(a.id)}>
                      {a.name}
                    </SelectItem>
                  ))}
                </SelectContent>
              </Select>
            </div>

            <div className="space-y-2">
              <Label>Menge</Label>
              <Input
                type="number"
                min={1}
                value={form.quantity}
                onChange={e =>
                  setForm(f => ({ ...f, quantity: Number(e.target.value) }))
                }
                className="bg-secondary border-border"
              />
            </div>

            <div className="space-y-2">
              <Label>Status</Label>
              <Select
                value={form.status}
                onValueChange={v =>
                  setForm(f => ({ ...f, status: v as OrderStatus }))
                }
              >
                <SelectTrigger className="bg-secondary border-border">
                  <SelectValue />
                </SelectTrigger>
                <SelectContent className="bg-card border-border">
                  {ORDER_STATUSES.map(s => (
                    <SelectItem key={s} value={s}>
                      {STATUS_LABELS[s]}
                    </SelectItem>
                  ))}
                </SelectContent>
              </Select>
            </div>
          </div>

          <DialogFooter>
            <Button
              variant="ghost"
              onClick={() => setDialogOpen(false)}
              className="text-muted-foreground"
            >
              Abbrechen
            </Button>
            <Button
              onClick={handleSubmit}
              className="bg-primary text-primary-foreground hover:bg-primary/90"
            >
              {editingOrder ? 'Speichern' : 'Erstellen'}
            </Button>
          </DialogFooter>
        </DialogContent>
      </Dialog>
    </div>
  )
}
```
